  /* ===== Myみっちー ===== */
  var MYMITCHIE_KEY = "mitchieMyCity";
  var registerMode = false;

  function shortLabel(l) {
    return l.replace("みっちー", "").replace(/な$/, "");
  }

  function getMyCity() {
    try { return localStorage.getItem(MYMITCHIE_KEY) || ""; } catch (e) { return ""; }
  }
  function setMyCity(name) {
    try { localStorage.setItem(MYMITCHIE_KEY, name); } catch (e) {}
  }

  /* 全自治体を点数順に並べて順位を求める。
     市区町村と都道府県は別の計算式なので、混ぜずに分けて数える。 */
  function buildRanks() {
    var muni = [], pref = [];
    for (var k in DB) {
      var e = DB[k];
      if (!e) continue;
      var isP = (e.p === k);
      var s = calcH(e.f, e.d, e.x, e.u, e.r, e.eo, isP, e.sfs);
      (isP ? pref : muni).push({name: k, pref: e.p, score: s});
    }
    var byScore = function(a, b){ return b.score - a.score; };
    muni.sort(byScore); pref.sort(byScore);
    var R = {muniRank: {}, muniTotal: muni.length,
             prefRank: {}, prefTotal: pref.length,
             inPref: {}, inPrefTotal: {}, score: {}};
    muni.forEach(function(o, i){ R.muniRank[o.name] = i + 1; R.score[o.name] = o.score; });
    pref.forEach(function(o, i){ R.prefRank[o.name] = i + 1; R.score[o.name] = o.score; });
    var groups = {};
    muni.forEach(function(o){ (groups[o.pref] = groups[o.pref] || []).push(o); });
    for (var p in groups) {
      R.inPrefTotal[p] = groups[p].length;
      groups[p].forEach(function(o, i){ R.inPref[o.name] = i + 1; });
    }
    return R;
  }

  var _ranks = null;
  function ranks() { if (!_ranks) _ranks = buildRanks(); return _ranks; }

  /* 順位を帯で示す。右へ行くほど上位。
     目印はみっちーの絵。端でも枠からはみ出さないよう位置を丸める。 */
  function rankBar(rank, total) {
    var pos = total > 1 ? (total - rank) / (total - 1) * 100 : 50;
    if (pos < 3) pos = 3;
    if (pos > 97) pos = 97;
    var img = (typeof IMGS !== "undefined" && IMGS.top)
      ? "<img class='mm-bar-pin' src='data:image/png;base64," + IMGS.top +
        "' alt='' style='left:" + pos.toFixed(1) + "%;'>"
      : "<div class='mm-bar-pin mm-bar-pin-fb' style='left:" + pos.toFixed(1) + "%;'></div>";
    return "<div class='mm-bar'><div class='mm-bar-track'></div>" + img + "</div>" +
      "<div class='mm-bar-lab'><span>" + total.toLocaleString() + "位</span>" +
      "<span>1位</span></div>";
  }

  /* カード右上の閉じるボタン。開閉はこれ一本にまとめている。 */
  function closeBtnHtml() {
    return "<button class='mm-close' id='mmCloseBtn' aria-label='Myみっちーを閉じる'>&#10005;</button>";
  }

  function renderMyMitchie() {
    var body = document.getElementById("myMitchieBody");
    if (!body || !DB) return;
    var name = getMyCity();
    var e = name ? DB[name] : null;

    if (!e) {
      body.innerHTML = closeBtnHtml() +
        "<div class='mm-empty'>" +
          "<p>住んでいる町を登録すると、<br>いつでもすぐ見られるよ</p>" +
          "<button class='mm-reg-btn' id='mmRegBtn'>Myみっちーを登録する</button>" +
          "<p class='mm-hint'>押すと検索バーに入力できるよ</p>" +
        "</div>";
      var rb = document.getElementById("mmRegBtn");
      if (rb) rb.addEventListener("click", startRegister);
      bindClose();
      return;
    }

    var R = ranks();
    var isP = (e.p === name);
    var sc = R.score[name];
    var pr = prof(sc);
    var html = closeBtnHtml() +
      "<div class='mm-head'><span class='mm-name'>" + name + "</span>" +
      "<span class='mm-score' style='color:" + pr.c + ";'>" + sc + "</span>" +
      "<span class='mm-badge' style='background:" + pr.c + "22;color:" + pr.c + ";'>" + shortLabel(pr.l) + "</span></div>";

    if (isP) {
      html += "<p class='mm-lab'>全国では <strong>" + R.prefRank[name] + "位</strong> / " + R.prefTotal + "都道府県</p>";
      html += rankBar(R.prefRank[name], R.prefTotal);
    } else {
      html += "<p class='mm-lab'>全国では <strong>" + R.muniRank[name].toLocaleString() + "位</strong> / " + R.muniTotal.toLocaleString() + "自治体</p>";
      html += rankBar(R.muniRank[name], R.muniTotal);
      html += "<p class='mm-lab'>" + e.p + "内では <strong>" + R.inPref[name] + "位</strong> / " + R.inPrefTotal[e.p] + "市町村</p>";
      html += rankBar(R.inPref[name], R.inPrefTotal[e.p]);
      var pe = DB[e.p];
      if (pe) {
        var ps = R.score[e.p], ppr = prof(ps);
        html += "<div class='mm-sep'></div>";
        html += "<div class='mm-head'><span class='mm-name mm-name-s'>" + e.p + "</span>" +
          "<span class='mm-score mm-score-s' style='color:" + ppr.c + ";'>" + ps + "</span>" +
          "<span class='mm-badge' style='background:" + ppr.c + "22;color:" + ppr.c + ";'>" + pshortLabel(pr.l) + "</span></div>";
        html += "<p class='mm-lab'>全国では <strong>" + R.prefRank[e.p] + "位</strong> / " + R.prefTotal + "都道府県</p>";
        html += rankBar(R.prefRank[e.p], R.prefTotal);
      }
    }

    html += "<div class='mm-actions'>" +
      "<button class='mm-go' id='mmGoBtn'>くわしく見る</button>" +
      "<button class='mm-edit' id='mmEditBtn' aria-label='Myみっちーを変更する'>&#9881;</button></div>";
    body.innerHTML = html;

    var g = document.getElementById("mmGoBtn");
    if (g) g.addEventListener("click", function(){
      document.getElementById("cityInput").value = name;
      hideSuggestions(); diagnose();
    });
    var ed = document.getElementById("mmEditBtn");
    if (ed) ed.addEventListener("click", startRegister);
    bindClose();
  }

  /* 登録モードに入る。検索バーへ誘導する。 */
  function startRegister() {
    registerMode = true;
    var ci = document.getElementById("cityInput");
    if (!ci) return;
    ci.value = "";
    ci.placeholder = "登録したい自治体名を入力";
    ci.classList.add("registering");
    ci.focus();
    ci.scrollIntoView({block: "center"});
  }

  function endRegister() {
    registerMode = false;
    var ci = document.getElementById("cityInput");
    if (!ci) return;
    ci.placeholder = "都道府県名または市区町村名を入力";
    ci.classList.remove("registering", "typing");
  }

  function bindClose() {
    var cb = document.getElementById("mmCloseBtn");
    if (cb) cb.addEventListener("click", toggleMyMitchie);
  }

  /* ボタンの先頭アイコンにみっちーの絵を入れる */
  function setMyMitchieIcon() {
    var ic = document.getElementById("mmIcon");
    if (ic && typeof IMGS !== "undefined" && IMGS.top) {
      ic.src = "data:image/png;base64," + IMGS.top;
    }
  }

  function toggleMyMitchie() {
    var box = document.getElementById("myMitchieBox");
    var btn = document.getElementById("myMitchieBtn");
    if (!box || !btn) return;
    var closed = box.classList.toggle("hidden");
    btn.setAttribute("aria-expanded", closed ? "false" : "true");
    /* 閉じているとき：みっちーの絵 ＋「Myみっちー ▼」
       開いているとき：「閉じる ▲」だけ */
    /* 開いているときはカード右上の×で閉じるので、このボタンは隠す */
    btn.style.display = closed ? "" : "none";
    if (!closed) renderMyMitchie();
  }
